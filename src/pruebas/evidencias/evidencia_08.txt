<soap:Envelope xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/" xmlns:ser="urn:MotocicletasService">

  <soap:Body>
    <ser:actualizarMotocicleta>
      <id_motocicleta>7</id_motocicleta>
      <marca>KTM</marca><modelo>Duke 200 NG</modelo><anio>2025</anio><cilindraje>200</cilindraje>

  </soap:Body>
</soap:Envelope>

<?xml version="1.0" ?>
